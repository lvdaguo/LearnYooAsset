## 接入 YooAsset 的完整步骤

### 1. 安装包

**方式 A（推荐）**：通过 OpenUPM 安装

在 `Packages/manifest.json` 中添加：

```json
{
  "scopedRegistries": [
    {
      "name": "package.openupm.com",
      "url": "https://package.openupm.com",
      "scopes": ["com.tuyoogame.yooasset"]
    }
  ],
  "dependencies": {
    "com.tuyoogame.yooasset": "2.3.18"
  }
}
```

**方式 B**：直接放到 `Assets/YooAsset/`（本项目就是这种方式）

### 3. 游戏初始化模板

创建一个 `GameInit.cs` 挂到启动场景的永久 GameObject：

```csharp
using System.Collections;
using UnityEngine;
using YooAsset;

public class GameInit : MonoBehaviour
{
    public EPlayMode PlayMode = EPlayMode.EditorSimulateMode;

    private IEnumerator Start()
    {
        DontDestroyOnLoad(gameObject);

        // 初始化 YooAsset
        YooAssets.Initialize();

        // 初始化包
        var package = YooAssets.CreatePackage("DefaultPackage");
        yield return InitPackage(package);

        // 热更新（仅 HostPlayMode 需要）
        if (PlayMode == EPlayMode.HostPlayMode)
            yield return PatchUpdate(package);

        // 设置默认包
        YooAssets.SetDefaultPackage(package);

        // 进入游戏主场景
        LoadMainScene();
    }

    private IEnumerator InitPackage(ResourcePackage package)
    {
        InitializationOperation initOp;

        switch (PlayMode)
        {
            case EPlayMode.EditorSimulateMode:
#if UNITY_EDITOR
                var buildResult = EditorSimulateModeHelper.SimulateBuild("DefaultPackage");
                var ep = new EditorSimulateModeParameters();
                ep.EditorFileSystemParameters =
                    FileSystemParameters.CreateDefaultEditorFileSystemParameters(buildResult.PackageRootDirectory);
                initOp = package.InitializeAsync(ep);
#else
                yield break;
#endif
                break;

            case EPlayMode.OfflinePlayMode:
                var op = new OfflinePlayModeParameters();
                op.BuildinFileSystemParameters =
                    FileSystemParameters.CreateDefaultBuildinFileSystemParameters();
                initOp = package.InitializeAsync(op);
                break;

            default: // HostPlayMode
                var hp = new HostPlayModeParameters();
                hp.BuildinFileSystemParameters =
                    FileSystemParameters.CreateDefaultBuildinFileSystemParameters();
                hp.CacheFileSystemParameters =
                    FileSystemParameters.CreateDefaultCacheFileSystemParameters(new RemoteServices());
                initOp = package.InitializeAsync(hp);
                break;
        }

        yield return initOp;
        if (initOp.Status != EOperationStatus.Succeed)
            Debug.LogError($"YooAsset 初始化失败: {initOp.Error}");
    }

    private IEnumerator PatchUpdate(ResourcePackage package)
    {
        // 拉取版本
        var versionOp = package.RequestPackageVersionAsync();
        yield return versionOp;
        if (versionOp.Status != EOperationStatus.Succeed) yield break;

        // 更新清单
        var manifestOp = package.UpdatePackageManifestAsync(versionOp.PackageVersion);
        yield return manifestOp;
        if (manifestOp.Status != EOperationStatus.Succeed) yield break;

        // 下载
        var downloader = package.CreateResourceDownloader(10, 3);
        if (downloader.TotalDownloadCount > 0)
        {
            downloader.DownloadUpdateCallback = OnDownloadUpdate;
            downloader.BeginDownload();
            yield return downloader;
        }

        // 清理旧缓存
        yield return package.ClearCacheFilesAsync(EFileClearMode.ClearUnusedBundleFiles);
    }

    private void OnDownloadUpdate(DownloadUpdateData data)
    {
        Debug.Log($"下载进度: {data.Progress:P0} ({data.CurrentDownloadCount}/{data.TotalDownloadCount})");
    }

    private void LoadMainScene() { /* SceneManager.LoadScene("MainMenu"); */ }
}

public class RemoteServices : IRemoteServices
{
    private const string CDN = "http://your-cdn.com/v1.0";
    public string GetRemoteMainURL(string fileName)     => $"{CDN}/{fileName}";
    public string GetRemoteFallbackURL(string fileName) => $"{CDN}/{fileName}";
}
```

### 4. 资源管理最佳实践

#### 方案 A：MonoBehaviour 级别管理（简单场景）

```csharp
public class EnemySpawner : MonoBehaviour
{
    private AssetHandle _enemyHandle;

    private IEnumerator Start()
    {
        _enemyHandle = YooAssets.LoadAssetAsync<GameObject>("enemy_normal");
        yield return _enemyHandle;
    }

    public void SpawnEnemy(Vector3 pos)
    {
        _enemyHandle.InstantiateSync(pos, Quaternion.identity);
    }

    private void OnDestroy()
    {
        _enemyHandle?.Release();
    }
}
```

#### 方案 B：Handle 池管理（高频实例化）

```csharp
// 战斗房间、关卡管理器等高频场景
public class LevelManager : MonoBehaviour
{
    private readonly List<AssetHandle> _handles = new List<AssetHandle>();

    public void LoadEntity(string address, Vector3 pos)
    {
        var handle = YooAssets.LoadAssetAsync<GameObject>(address);
        handle.Completed += h => h.InstantiateSync(pos, Quaternion.identity);
        _handles.Add(handle);
    }

    private void OnDestroy()
    {
        foreach (var h in _handles) h.Release();
        _handles.Clear();
        StartCoroutine(UnloadUnused());
    }

    private IEnumerator UnloadUnused()
    {
        var package = YooAssets.GetPackage("DefaultPackage");
        yield return package.UnloadUnusedAssetsAsync();
    }
}
```

#### 方案 C：静态资源缓存管理器

```csharp
// 常驻资源（UI皮肤、公共图集、音效）
public static class AssetCache
{
    private static readonly Dictionary<string, AssetHandle> _cache = new();

    public static IEnumerator Preload(params string[] addresses)
    {
        foreach (var addr in addresses)
        {
            if (_cache.ContainsKey(addr)) continue;
            var handle = YooAssets.LoadAssetAsync<UnityEngine.Object>(addr);
            yield return handle;
            _cache[addr] = handle;
        }
    }

    public static T Get<T>(string address) where T : UnityEngine.Object
        => _cache.TryGetValue(address, out var h) ? h.GetAssetObject<T>() : null;

    public static void Release(string address)
    {
        if (_cache.TryGetValue(address, out var h))
        {
            h.Release();
            _cache.Remove(address);
        }
    }
}
```

### 5. CI/CD 自动构建配置

```csharp
// Editor/BuildScript.cs
using UnityEditor;
using YooAsset.Editor;

public static class BuildScript
{
    // Jenkins / GitHub Actions 命令行调用：
    // Unity.exe -batchmode -executeMethod BuildScript.BuildAndroid -quit
    public static void BuildAndroid()
    {
        var param = new BuiltinBuildParameters
        {
            BuildOutputRoot       = System.Environment.GetEnvironmentVariable("BUILD_OUTPUT") ?? "D:/Output",
            BuildinFileRoot       = "Assets/StreamingAssets",
            BuildTarget           = BuildTarget.Android,
            PackageName           = "DefaultPackage",
            PackageVersion        = System.DateTime.Now.ToString("yyyyMMdd_HHmmss"),
            FileNameStyle         = EFileNameStyle.HashName,
            CompressOption        = ECompressOption.LZ4,
            BuildinFileCopyOption = EBuildinFileCopyOption.ClearAndCopyAll,
            UseAssetDependencyDB  = true,
            ClearBuildCacheFiles  = false,
        };

        var result = new BuiltinBuildPipeline().Run(param, enableLog: true);
        if (!result.Success)
        {
            Debug.LogError(result.ErrorInfo);
            EditorApplication.Exit(1);
        }
        Debug.Log($"构建成功: {result.OutputPackageDirectory}");
    }
}
```

### 6. 版本管理建议

| 场景 | PackageVersion 格式 | 说明 |
|------|-------------------|------|
| 每日构建 | `20250110_143022` | 时间戳，唯一性强 |
| 正式发布 | `1.2.3` | 语义化版本 |
| 热更新补丁 | `1.2.3.patch.001` | 在大版本基础上叠加 |

服务器目录结构要对应版本号，CDN 保留多个版本以支持**版本回退**。

### 7. 常见坑与解决方案

| 问题 | 原因 | 解决 |
|------|------|------|
| 编辑器下资源找不到 | Collector 未配置此路径 | 检查 Collector 路径和 Active 状态 |
| 真机加载失败，编辑器正常 | EditorSimulate 未打包就发布 | 发布前必须用 BuiltinBuildPipeline 构建 |
| 热更新后资源版本不对 | CDN 路径与 PackageVersion 不匹配 | 确认上传目录结构 |
| 内存持续增长 | Handle 未 Release | 每个 Handle 都要对应 Release |
| Bundle 卸载后资源失效 | AutoUnloadBundleWhenUnused=true 时 Handle 被错误回收 | 保持 Handle 引用直到不再需要 |
| 场景切换后旧资源残留 | 忘记调用 UnloadUnusedAssetsAsync | 在场景卸载事件中调用 |

### 8. 推荐的项目目录结构

```
Assets/
├── _YooAssetConfig/
│   └── AssetBundleCollectorSetting.asset   ← Collector 配置文件
├── StreamingAssets/                        ← 首包 Bundle（构建时自动填充）
├── Resources/
│   └── PatchWindow.prefab                  ← 热更新 UI（唯一走 Resources 的资源）
└── GameAssets/
    ├── UI/
    ├── Characters/
    ├── Scenes/
    ├── Audio/
    └── Config/                             ← JSON / 二进制配置（RawFile）
```
